🔍 Search System — Документация

# Компонент поиска с поддержкой локальной фильтрации, глобального поиска через API и гибридного режима. Построен на React + TypeScript с фокусом на производительность, типобезопасность и доступность.

✨ Возможности

Фича ||| Описание
🔹 Локальный поиск | Мгновенная фильтрация уже загруженных данных на клиенте

🔹 Глобальный поиск | Поиск через бэкенд с пагинацией и дебаунсом

🔹 Гибридный режим | Автопереключение: @запрос → бэкенд, запрос → локально

🔹 Типобезопасность | Дженерики <T> + строгие ограничения типов

🔹 Доступность | ARIA-атрибуты, клавиша Escape, семантическая разметка

🔹 Оптимизация | React.memo, useCallback, useMemo, AbortController

🔹 Гибкость | Проброс ...inputProps, кастомные рефы, стилизация через className

🧩 Компонент Search

Базовый UI-компонент: инпут с иконкой, кнопкой очистки и обработкой клавиш.

🔧 Props

interface SearchProps extends Omit<React.InputHTMLAttributes<HTMLInputElement>,'onChange' | 'ref'

> {
> className?: string; // Класс для контейнера
> inputClassName?: string; // Класс для <input>
> value: string; // Контролируемое значение
> onChange: (value: string) => void; // Колбэк на изменение
> placeholder?: string; // Подсказка (default: 'Поиск...')
> inputRef?: React.Ref<HTMLInputElement>; // Реф для инпута
> disableClear?: boolean; // Скрыть кнопку очистки (default: false)
> showIcon?: boolean; // Показать иконку поиска (default: true)
> onClear?: () => void; // Колбэк при очистке / Escape
> }

🔧 Функции фильтрации

# Поиск по чатам: name, chat.username, chat.nickname, last_message.content и др.

filterChatsLocal<T>(items, searchTerm)

// Тип входных данных
type Chat = {
name: string;
chat: {
username?: string | null;
nickname?: string | null;
first_name?: string | null;
last_name?: string | null;
patronymic?: string | null;
};
last_message?: { content?: string | null } | null;
};

// Использование
const results = filterChatsLocal(chats, 'John');

# Поиск по контактам: first_name, last_name, nickname, username.

filterContacts<T>(items, searchTerm)

const results = filterContacts(contacts, 'петров');

# Универсальный поиск по одному полю.

filterByName<T>(items, searchTerm, fieldName?)

// Поиск по полю 'username'

const results = filterByName(users, 'dev', 'username');

# Поиск по нескольким полям объекта.

filterByNameExtended<T>(items, searchTerm, fields?)

// Поиск по nickname, first_name, last_name
const results = filterByNameExtended(users, 'иван', [
'nickname', 'first_name', 'last_name'
]);

# Низкоуровневый фильтр по одному полю (возвращает [] при пустом запросе).

filterByField<T>(items, searchTerm, field)

const results = filterByField(items, 'test', 'email');

🪝 Хуки поиска

useLocalSearch<T>

Назначение: Локальная фильтрация массива на клиенте.

const {
searchTerm, // Текущее значение инпута
filteredData, // Отфильтрованный массив
handleSearchChange, // Обработчик onChange
handleClear // Обработчик очистки
} = useLocalSearch(
data, // T[] — исходные данные
filterFn, // (items: T[], term: string) => T[]
300 // debounceDelay (опционально)
);

# Пример: поиск по чатам

const { searchTerm, filteredData, handleSearchChange } = useLocalSearch(
chats,
filterChatsLocal,
200
);

return (
<>
<Search value={searchTerm} onChange={handleSearchChange} />
<List items={filteredData} />
</>
);

useGlobalSearch<T>
Назначение: Поиск через бэкенд с дебаунсом и обработкой загрузки.

const {
searchTerm, // Значение инпута
results, // Результаты от API
isLoading, // Индикатор загрузки
error, // Ошибка запроса
handleSearchChange,
handleClear
} = useGlobalSearch(
async (term: string) => {
const res = await api.searchChats(term);
return res.data;
},
500 // debounceDelay
);

# Пример: глобальный поиск пользователей

const { results, isLoading, handleSearchChange } = useGlobalSearch(
async (query) => {
const response = await fetch(`/api/users?q=${query}`);
return response.json();
}
);

return (
<>
<Search value={searchTerm} onChange={handleSearchChange} />
{isLoading && <Spinner />}
<UserList users={results} />
</>
);

const onSearch = async (term: string, signal?: AbortSignal) => {
const res = await fetch(`/api/search?q=${term}`, { signal });
return res.json();
};

# Назначение: Комбинированный поиск с автопереключением по префиксу @.

useHybridSearch<T> ⭐

Ввод | Режим | Источник данных
JOHN | 🔹 Локальный | localData + localFilterFn
@JOHN | 🔹 Глобальный | globalSearchFn (бэкенд)

const {
searchTerm, // Значение инпута
results, // Актуальные результаты (local или global)
globalResults, // Только результаты бэкенда
localResults, // Только локально отфильтрованные
isGlobal, // Флаг: активен ли глобальный режим
isLoading,
error,
handleSearchChange,
handleClear,
handleClearLocal, // Очистить только локальные
handleClearGlobal // Очистить только глобальные + отменить запрос
} = useHybridSearch(
localData, // T[] — данные для локального поиска
localFilterFn, // (items, term) => T[]
async (term, signal) => {
// Глобальный поиск: поддерживаем AbortSignal
const res = await api.search(term, { signal });
return res.data;
},
300, // debounceDelay
'@' // globalPrefix
);

🔹 Сценарий 4: ChatList с fallback-логикой (production-ready)
const ChatList = ({ selectedChatUid }) => {
const [searchQuery, setSearchQuery] = useState('');
const [debouncedQuery, setDebouncedQuery] = useState('');

const debouncedSetQuery = useDebounce(setDebouncedQuery, 300);
useEffect(() => { debouncedSetQuery(searchQuery); }, [searchQuery]);

const isGlobalMode = debouncedQuery?.trim().startsWith('@');
const searchValue = isGlobalMode
? debouncedQuery.trim().slice(1).trim()
: debouncedQuery?.trim() || '';

// Бэкенд-запрос только в глобальном режиме
const { serverResponse } = useGetChatsQuery(
{ search: isGlobalMode ? searchValue : undefined, pageSize: 50 },
{ skip: !isGlobalMode || !searchValue }
);

// Локальная фильтрация только в локальном режиме
const filteredLocal = useMemo(() => {
if (isGlobalMode || !searchValue) return mockChats;
return filterChatsLocal(mockChats, searchValue);
}, [isGlobalMode, searchValue]);

// Выбор результатов
const displayChats = isGlobalMode
? (serverResponse?.results ?? [])
: filteredLocal;

return (
<>
<Search
value={searchQuery}
onChange={setSearchQuery}
placeholder={isGlobalMode ? 'Глобальный поиск...' : 'Поиск чатов...'}
/>
<List items={displayChats} selectedId={selectedChatUid} />
</>
);
};

# Оптимизации "из коробки"

// ✅ React.memo — предотвращает перерендер при неизменных пропсах
export const Search = memo(forwardRef(...));

// ✅ useCallback — стабильные ссылки на обработчики
const handleChange = useCallback((e) => onChange(e.target.value), [onChange]);

// ✅ useMemo — кэширование вычисляемых классов/значений
const containerClass = useMemo(() => classNames(cls.root, className), [className]);

// ✅ useDebounce — уменьшение частоты вызовов фильтрации/API
const debouncedFn = useDebounce(callback, 300);

// ✅ AbortController — отмена устаревших запросов при быстром вводе
const controller = new AbortController();
fetch(url, { signal: controller.signal });

# Рекомендации

Не передавайте новые функции/объекты в пропсы без useCallback/useMemo:

// ❌ Плохо: создаёт новую функцию каждый рендер
<Search onChange={(v) => setQuery(v)} />

// ✅ Хорошо: стабильная ссылка
const handleChange = useCallback((v) => setQuery(v), []);
<Search onChange={handleChange} />

## Ошибки

❌ "Cannot read properties of undefined (reading 'trim')"

Причина: Попытка вызвать .trim() на undefined (обычно debouncedQuery на первом рендере).

Решение:
// ✅ Используйте optional chaining
const query = debouncedQuery?.trim();
if (!query) return items;

// ✅ Или инициализируйте стейт пустой строкой
const [debouncedQuery, setDebouncedQuery] = useState('');

Локальный поиск не фильтрует
Чеклист:

// 1. Проверьте, что debouncedQuery обновляется
useEffect(() => {
console.log('debouncedQuery:', debouncedQuery);
}, [debouncedQuery]);

// 2. Убедитесь, что фильтр получает правильные данные
console.log('mockChats[0]:', mockChats[0]);

// 3. Проверьте, что поля в фильтре совпадают со структурой данных
// В filterChatsLocal ожидаются: name, chat.username, last_message.content

// 4. Убедитесь, что не срабатывает приоритет бэкенда
// Если serverResponse имеет результаты, они перекрывают локальные

❌ Типы TypeScript: "Type 'Chat' is not assignable to 'Record<string, unknown>'"
Причина: Функция фильтрации ожидает Record<string, unknown>, но получает конкретный тип.
Решение:
// ✅ Используйте конкретный тип в дженерике фильтра
export function filterChatsLocal<
T extends {
name: string;
chat: { username?: string | null; /_ ... _/ };
last_message?: { content?: string | null } | null;
}

> (items: T[], searchTerm: string): T[] { ... }

// ✅ Или импортируйте тип из API
import type { Chat } from '@/entities/Chat';
export function filterChatsLocal(items: Chat[], searchTerm: string): Chat[] { ... }

❌ Запросы не отменяются при быстром вводе
Решение: Передавайте AbortSignal в API-функцию:

// В useHybridSearch / useGlobalSearch
const data = await globalSearchFn(term, controller.signal);

// В API-слое
export const searchChats = async (query: string, signal?: AbortSignal) => {
const res = await fetch(`/api/chats?q=${query}`, { signal });
return res.json();
};
